    Histogram readers return a Histogram; load_scatter returns ScatterData. ROOT and boost-histogram readers share the histogram type so their outputs are interchangeable downstream.

    ROOT backend (unrooted.io.root)

    Functions for reading ROOT files via uproot. No ROOT installation required.

    All four functions below are public exports of unrooted.io.root. load accepts TH1, TH2, and TProfile; load_efficiency accepts separate unweighted passed/total TH1s. TH3, TProfile2D, and direct TEfficiency loading are unsupported. Branch readers accept numeric scalars, vectors, and nested jagged arrays; paired branches must be broadcast-compatible. See selection and memory behavior.

    def load(path: str | Path, key: str) -> Histogram:
    """Load a ROOT TH1, TH2, or TProfile as an independent array snapshot.

    Values, variances, numeric edges, axis labels, and both flow bins per axis
    are extracted through uproot. TProfile spread bounds are mean ± σ_y,
    derived as sqrt(SE² * N); empty bins have NaN spread.

    Args:
        path: Path to the ROOT file.
        key: Object key, optionally including a cycle (e.g. ``hx;1``).

    Returns:
        A Histogram whose name is the key with its cycle stripped.

    Raises:
        KeyError: If the key is absent.
        ValueError: If the object type or its numerical data are unsupported.
    """
    with cast(Any, uproot.open(path)) as f:
        # Inspect metadata before deserializing unsupported objects such as
        # TEfficiency, which uproot may be unable to decode at all.
        _check_class(f.classname_of(key), key)
        return _from_root(f[key], key)

    def load_efficiency(
    path: str | Path,
    pass_key: str,
    total_key: str,
    *,
    confidence_level: float = DEFAULT_CONFIDENCE_LEVEL,
) -> Histogram:
    """Compute a binomial efficiency from separate passed and total TH1 objects.

    Requires 1D unweighted, finite, nonnegative integer counts with passed <=
    total, Poisson-compatible variances, and exactly identical edges. Profiles,
    TH2, and TEfficiency objects are unsupported.

    Values are passed/total, variances retain p*(1-p)/total, and spread bounds
    are Wilson intervals without continuity correction (68.268949% by default).
    All four arrays contain NaN for bins with zero total. Endpoint intervals
    have nonzero width, even though the variance field is zero there.

    Args:
        path: Path to the ROOT file.
        pass_key: Key counting passing events.
        total_key: Key counting total events.
        confidence_level: Nominal Wilson confidence level, strictly between 0 and 1.

    Raises:
        KeyError: If either key is absent.
        ValueError: If inputs, binning, or confidence level are unsupported.
    """
    with cast(Any, uproot.open(path)) as f:
        if any(
            not f.classname_of(key).startswith("TH1") for key in (pass_key, total_key)
        ):
            raise ValueError(
                "Efficiency inputs must be one-dimensional TH1 count histograms"
            )
        return binomial_efficiency(
            _from_root(f[pass_key], pass_key),
            _from_root(f[total_key], total_key),
            confidence_level=confidence_level,
        )

    def load_branch(
    path: str | Path,
    key: str,
    x_branch: str,
    y_branch: str | None = None,
    *,
    n_bins: int = 100,
    range: tuple[float, float] | None = None,  # noqa: A002
    label: str = "",
    entry_start: int = 0,
    entry_stop: int | None = None,
    step_size: int = DEFAULT_STEP_SIZE,
) -> Histogram:
    """Load selected tree entries as a count or population-profile histogram.

    Scalars, vectors, and nested jagged branches are flattened. Profile pairs
    are aligned with awkward.broadcast_arrays within each chunk. Bins are
    [left, right), including the final right edge. Out-of-range x is omitted.

    Count histograms need one accumulation pass. Profiles use three passes:
    observed extrema/counts, sums shifted by the global per-bin minima, and
    centered squared deviations. This preserves numerical stability at large
    offsets without retaining all observations. Automatic ranges require an
    additional x-only pass before broadcasting.

    Profile variance is M2/N². Empty bins have value/variance zero and NaN
    extrema; single-entry bins have zero variance and equal extrema.

    Args:
        path: ROOT file path.
        key: Tree path inside the file.
        x_branch: Branch defining the bin axis.
        y_branch: Optional branch to profile against x.
        n_bins: Positive number of bins (default 100).
        range: Finite (lo, hi), lo < hi; None scans the selected x values.
        label: X-axis label; defaults to x_branch.
        entry_start: First entry included, nonnegative (default zero).
        entry_stop: First entry excluded; None reads to the end. Bounds beyond
                    the tree length are clipped. Selection precedes flattening.
        step_size: Maximum entries per chunk (default 100,000). This is not a
                   byte limit; jagged expansion and ROOT baskets affect memory.
    """
    _validate_entries(entry_start, entry_stop)
    _positive_integer(step_size, "step_size")
    _positive_integer(n_bins, "n_bins")
    with cast(Any, uproot.open)(path, array_cache=None) as root_file:
        tree = _get_tree(root_file, key, x_branch, y_branch)
        bounds = _range(tree, x_branch, range, entry_start, entry_stop, step_size)
        axis = Axis(np.linspace(*bounds, int(n_bins) + 1), label=label or x_branch)
        if y_branch is None:
            counts = np.zeros(axis.n_bins, dtype=np.int64)
            for x, _ in _chunks(
                tree, x_branch, None, entry_start, entry_stop, step_size
            ):
                counts += np.histogram(x, bins=axis.edges)[0]
            values = counts.astype(float)
            return Histogram([axis], values, values.copy(), name=x_branch)
        return _profile(
            tree, x_branch, y_branch, axis, entry_start, entry_stop, step_size
        )

    def load_scatter(
    path: str | Path,
    key: str,
    x_branch: str,
    y_branch: str,
    *,
    label_x: str = "",
    label_y: str = "",
    entry_start: int = 0,
    entry_stop: int | None = None,
    step_size: int = DEFAULT_STEP_SIZE,
    max_points: int | None = None,
    sampling: str = "error",
    seed: int = 0,
) -> ScatterData:
    """Load point pairs in chunks, with an optional explicit limit or sample.

    Entry selection precedes awkward broadcasting and flattening. Without
    max_points all selected pairs are returned, so result memory grows with
    the selection. No sampling happens implicitly.

    Args:
        path: ROOT file path.
        key: Tree path inside the file.
        x_branch: Branch for x coordinates.
        y_branch: Branch for y coordinates.
        label_x: X-axis label; defaults to x_branch.
        label_y: Y-axis label; defaults to y_branch.
        entry_start: First entry included, nonnegative (default zero).
        entry_stop: First entry excluded; None reads to the end. Bounds beyond
                    the tree length are clipped.
        step_size: Maximum entries per chunk, not bytes (default 100,000).
        max_points: Positive maximum retained pairs, or None for all pairs.
        sampling: 'error' raises when max_points is exceeded. 'random' retains
                  a uniform sample without replacement over flattened pairs;
                  requires max_points. Returned points keep their input order.
        seed: Nonnegative RNG seed (default zero), used by random sampling.
              Results repeat for the same data/seed across chunk sizes.
    """
    _validate_entries(entry_start, entry_stop)
    _positive_integer(step_size, "step_size")
    if max_points is not None:
        _positive_integer(max_points, "max_points")
    if sampling not in ("error", "random"):
        raise ValueError("sampling must be 'error' or 'random'")
    if sampling == "random" and max_points is None:
        raise ValueError("sampling='random' requires max_points")
    if isinstance(seed, bool) or not isinstance(seed, Integral) or seed < 0:
        raise ValueError("seed must be a nonnegative integer")
    if sampling != "random" and seed != 0:
        raise ValueError("seed is only used with sampling='random'")
    accumulator = ScatterAccumulator(max_points, sampling, int(seed))
    with cast(Any, uproot.open)(path, array_cache=None) as root_file:
        tree = _get_tree(root_file, key, x_branch, y_branch)
        for x, y in _chunks(
            tree, x_branch, y_branch, entry_start, entry_stop, step_size
        ):
            assert y is not None
            accumulator.add(x, y)
    x, y = accumulator.finish()
    return ScatterData(
        x=x,
        y=y,
        name=y_branch,
        x_label=label_x or x_branch,
        y_label=label_y or y_branch,
        source_points=accumulator.total,
        sample_seed=int(seed) if sampling == "random" else None,
    )

    def load(hist: bh.Histogram, name: str = "") -> Histogram:
    """Convert a boost-histogram :class:`~boost_histogram.Histogram` to an
    unrooted :class:`~unrooted.core.Histogram`.

    Supports numeric ``Regular``, ``Variable``, and ``Integer`` axes and the
    following storage types:

    * ``Mean`` — treated as a profile histogram; populates ``spread_min`` /
      ``spread_max`` as mean ± σ_y (standard deviation of individual
      measurements).  Empty bins (count = 0) get ``nan`` spread.
    * ``Weight`` — structured storage with ``value`` and ``variance`` fields.
    * ``Double``, ``Int64``, ``Unlimited``, ``AtomicInt64`` — Poisson assumption;
      ``variances`` equals ``values``.

    Categorical/Boolean axes and ``WeightedMean`` storage are unsupported.
    Arrays are independent snapshots of the source. Missing flow slots are
    padded with NaN. Axis metadata strings become axis labels.

    Args:
        hist: The boost-histogram histogram to convert.
        name: Name assigned to the returned histogram.

    Returns:
        A :class:`~unrooted.core.Histogram` populated from *hist*.
    """
    if hist.ndim == 0:
        raise ValueError("Boost histogram must have at least one numeric axis")
    for ax in hist.axes:
        if not isinstance(ax, (bh.axis.Regular, bh.axis.Variable, bh.axis.Integer)):
            raise ValueError(
                f"Unsupported boost axis {type(ax).__name__}; "
                "use numeric Regular, Variable, or Integer axes"
            )
    if hist.storage_type not in (*_COUNT_STORAGES, bh.storage.Weight, bh.storage.Mean):
        raise ValueError(
            f"Unsupported boost storage {hist.storage_type.__name__}; "
            "use count, Weight, or Mean storage; WeightedMean is unsupported"
        )
    axes = [
        Axis(edges=np.array(ax.edges, copy=True), label=_axis_label(ax))
        for ax in hist.axes
    ]

    if hist.storage_type == bh.storage.Mean:
        view: Any = hist.view(flow=False)
        count = np.asarray(view["count"])
        values = np.array(view["value"], copy=True)
        sod2 = np.asarray(view["_sum_of_deltas_squared"])

        has_data = count > 0
        safe_count = np.where(count >= 2, count, 1.0)
        safe_count_m1 = np.where(count >= 2, count - 1, 1.0)

        # SE² = sod2 / (count * (count-1)); 0 for bins with 0 or 1 entries
        variances = np.where(
            count >= 2,
            np.maximum(0.0, sod2) / (safe_count * safe_count_m1),
            0.0,
        )

        # σ_y = std-dev of individual measurements; NaN for empty bins
        sigma_y = np.where(
            has_data,
            np.where(count >= 2, np.sqrt(np.maximum(0.0, sod2 / safe_count_m1)), 0.0),
            np.nan,
        )

        spread_min: np.ndarray | None = np.where(has_data, values - sigma_y, np.nan)
        spread_max: np.ndarray | None = np.where(has_data, values + sigma_y, np.nan)
        overflow_view: Any = hist.view(flow=True)
        overflow = _flow_values(hist, overflow_view["value"])

        return Histogram(
            axes=axes,
            values=values,
            variances=variances,
            name=name,
            overflow=overflow,
            spread_min=spread_min,
            spread_max=spread_max,
        )

    raw: Any = hist.view(flow=False)
    raw_flow: Any = hist.view(flow=True)
    if hist.storage_type == bh.storage.Weight:
        # Weight storage
        values = np.array(raw["value"], dtype=float, copy=True)
        variances = np.array(raw["variance"], dtype=float, copy=True)
        overflow = _flow_values(hist, raw_flow["value"])
    else:
        values = np.array(raw, dtype=float, copy=True)
        variances = values.copy()
        overflow = _flow_values(hist, raw_flow)

    return Histogram(
        axes=axes,
        values=values,
        variances=variances,
        name=name,
        overflow=overflow,
    )

    """Compute per-bin efficiency from two boost-histogram histograms.

    The returned :class:`~unrooted.core.Histogram` contains:

    * ``values``    — efficiency = accepted / total per bin (``nan`` for empty)
    * ``variances`` — binomial variance = eff × (1 − eff) / total

    The default confidence level is 68.268949%. Bins where *total* = 0

    unweighted, finite, nonnegative integer counts with accepted <= total
    and exactly identical edges are supported.

    Args:
        accepted: Histogram counting accepted (passing) events.
        total:    Histogram counting all (total) events.
        name:     Name assigned to the returned histogram.
        confidence_level: Nominal Wilson confidence level, strictly between 0 and 1.

    Returns:
        A :class:`~unrooted.core.Histogram` representing the efficiency.

    Raises:
        ValueError: If inputs, storage, binning, or confidence level are unsupported.
    """
    for hist in (accepted, total):
        if hist.storage_type not in _COUNT_STORAGES:
            raise ValueError(
                "Efficiency inputs require unweighted count storage "
                "(Double, Int64, Unlimited, or AtomicInt64); "
                "Weight, Mean, and WeightedMean are unsupported"
            )
        if hist.variances(flow=False) is None:
            raise ValueError(
                "Efficiency inputs require known unweighted count variances; "
                "weighted scalar fills are unsupported"
            )
    return binomial_efficiency(
        load(accepted),
        load(total),
        name=name,
        confidence_level=confidence_level,
    )
